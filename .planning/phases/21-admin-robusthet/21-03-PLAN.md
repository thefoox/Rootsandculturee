3. Update the `useEffect` to handle the not-found case and add a timeout:
```tsx
useEffect(() => {
  if (!params.id) return

  const timeout = setTimeout(() => {
    setLoading(false)
    setError('Tidsavbrudd — kunne ikke laste ordren. Prov igjen.')
  }, 15000) // 15 second timeout

  getOrderById(params.id).then((o) => {
    clearTimeout(timeout)
    if (!o) {
      setLoading(false)
      setError('Ordren ble ikke funnet.')
      return
    }
    setOrder(o)
    setLoading(false)
    setSelectedStatus(o.status)
    if (o.shipping) {
      setShippingForm(o.shipping)
    }
    if (o.stripePaymentIntentId) {
      getRefundsForOrder(o.stripePaymentIntentId).then(setRefunds).catch(() => toast.error('Kunne ikke laste refusjoner.'))
    }
  }).catch((err) => {
    clearTimeout(timeout)
    console.error('[OrderDetail] Failed to load order:', err)
    setLoading(false)
    setError('Kunne ikke laste ordredetaljer. Prov igjen.')
  })

  getOrderNotes(params.id).then(setNotes).catch(() => toast.error('Kunne ikke laste notater.'))

  return () => clearTimeout(timeout)
}, [params.id])
```
