**Del A — Slett-knapp (CMS-02):**

Legg til slett-knapp i tabellen og bruk DeleteConfirmDialog-mønsteret (som i `admin/produkter/page.tsx`).

1. Legg til imports øverst:
   ```typescript
   import { DeleteConfirmDialog } from '@/components/admin/DeleteConfirmDialog'
   import { toast } from 'sonner'
   ```

2. Legg til state:
   ```typescript
   const [deleteTarget, setDeleteTarget] = useState<PageContent | null>(null)
   const [isDeleting, setIsDeleting] = useState(false)
   ```

3. Legg til handleDelete-funksjon:
   ```typescript
   async function handleDelete() {
     if (!deleteTarget) return
     setIsDeleting(true)
     const res = await fetch(`/api/page-content/${deleteTarget.id}`, { method: 'DELETE' })
     if (res.ok) {
       setPages((prev) => prev.filter((p) => p.id !== deleteTarget.id))
       toast.success(`${deleteTarget.title} er slettet.`)
     } else {
       toast.error('Kunne ikke slette siden. Prøv igjen.')
     }
     setIsDeleting(false)
     setDeleteTarget(null)
   }
   ```

4. Legg til en ny kolonne i tabellen for handlinger — legg til etter "Sist oppdatert"-kolonnen:
   ```tsx
   <th className="px-5 py-3 text-label font-medium text-forest">Handlinger</th>
   ```
   Og i raden:
   ```tsx
   <td className="px-5 py-4">
     <button
       onClick={() => setDeleteTarget(page)}
       className="text-label text-body hover:text-destructive"
       aria-label={`Slett ${page.title}`}
     >
       Slett
     </button>
   </td>
   ```

5. Legg til DeleteConfirmDialog før avsluttende `</div>`:
   ```tsx
   <DeleteConfirmDialog
     isOpen={deleteTarget !== null}
     onClose={() => setDeleteTarget(null)}
     onConfirm={handleDelete}
     itemName={deleteTarget?.title || ''}
     isDeleting={isDeleting}
   />
   ```
