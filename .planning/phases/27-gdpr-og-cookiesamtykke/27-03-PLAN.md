Create `src/actions/account.ts` with two server actions:

```typescript
'use server'

import { verifySession } from '@/lib/dal'
import { adminDb } from '@/lib/firebase/admin'
import { deleteSession } from '@/lib/session'

/**
 * GDPR Art. 15 — Right of access / data export
 * Returns all personal data for the authenticated user as a JSON object.
 */
export async function exportUserData(): Promise<{
  success: boolean
  data?: object
  error?: string
}> {
  const session = await verifySession()
  if (!session) {
    return { success: false, error: 'Du er ikke logget inn.' }
  }

  const uid = session.uid

  try {
    const [userDoc, ordersSnap, bookingsSnap] = await Promise.all([
      adminDb.collection('users').doc(uid).get(),
      adminDb.collection('orders').where('customerId', '==', uid).get(),
      adminDb.collection('bookings').where('customerId', '==', uid).get(),
    ])

    const data = {
      profil: userDoc.exists ? userDoc.data() : null,
      ordrer: ordersSnap.docs.map((d) => ({ id: d.id, ...d.data() })),
      bookinger: bookingsSnap.docs.map((d) => ({ id: d.id, ...d.data() })),
      eksportertDato: new Date().toISOString(),
    }

    return { success: true, data }
  } catch {
    return {
      success: false,
      error: 'Kunne ikke hente dine data. Prov igjen.',
    }
  }
}

/**
 * GDPR Art. 17 — Right to erasure
 * Anonymizes orders/bookings (Bokforingsloven: 5 year retention),
 * deletes user profile document, and clears session.
 *
 * NOTE: Firebase Auth record deletion must happen CLIENT-SIDE
 * before calling this action, using getAuth().currentUser.delete().
 * adminAuth is null in this project (no Firebase Admin SDK on Vercel).
 */
export async function deleteUserAccount(): Promise<{
  success: boolean
  error?: string
}> {
  const session = await verifySession()
  if (!session) {
    return { success: false, error: 'Du er ikke logget inn.' }
  }

  const uid = session.uid

  try {
    // Fetch orders and bookings linked to this user
    const [ordersSnap, bookingsSnap] = await Promise.all([
      adminDb.collection('orders').where('customerId', '==', uid).get(),
      adminDb.collection('bookings').where('customerId', '==', uid).get(),
    ])

    const batch = adminDb.batch()

    // Anonymize orders — keep for Bokforingsloven (5 year retention)
    for (const doc of ordersSnap.docs) {
      batch.update(doc.ref, {
        customerId: null,
        customerEmail: '[slettet]',
        customerName: '[slettet]',
      })
    }

    // Anonymize bookings similarly
    for (const doc of bookingsSnap.docs) {
      batch.update(doc.ref, {
        customerId: null,
        customerEmail: '[slettet]',
        customerName: '[slettet]',
      })
    }

    // Delete user profile document
    batch.delete(adminDb.collection('users').doc(uid))

    await batch.commit()

    // Clear session cookie
    await deleteSession()

    return { success: true }
  } catch {
    return {
      success: false,
      error: 'Kunne ikke slette kontoen. Prov igjen eller kontakt oss.',
    }
  }
}
```

Key implementation details:
- `exportUserData` returns `{ success, data, error }` so the client can trigger a JSON download
- `deleteUserAccount` returns `{ success, error }` instead of using redirect() — the client needs to handle Firebase Auth deletion and then navigate
- Orders/bookings are anonymized (customerId: null, customerEmail: '[slettet]', customerName: '[slettet]') NOT deleted — per Bokforingsloven 5-year retention
- Firebase Auth record deletion is NOT done here because adminAuth is null. The client must call `user.delete()` from Firebase client SDK before this action
- Both actions use `verifySession()` guard at the top
- Error messages in Norwegian following the `'Kunne ikke [verb] [noun]. Prov igjen.'` pattern

```typescript
'use client'

import { useState } from 'react'
import { exportUserData } from '@/actions/account'
import { Button } from '@/components/ui/Button'
import { toast } from 'sonner'
import { Download } from 'lucide-react'

export function DataExportButton() {
  const [loading, setLoading] = useState(false)

  async function handleExport() {
    setLoading(true)
    try {
      const result = await exportUserData()
      if (!result.success || !result.data) {
        toast.error(result.error || 'Kunne ikke hente data.')
        return
      }

      // Trigger JSON file download
      const blob = new Blob([JSON.stringify(result.data, null, 2)], {
        type: 'application/json',
      })
      const url = URL.createObjectURL(blob)
      const a = document.createElement('a')
      a.href = url
      a.download = `mine-data-${new Date().toISOString().split('T')[0]}.json`
      document.body.appendChild(a)
      a.click()
      document.body.removeChild(a)
      URL.revokeObjectURL(url)
      toast.success('Data eksportert')
    } catch {
      toast.error('Noe gikk galt. Prov igjen.')
    } finally {
      setLoading(false)
    }
  }

  return (
    <Button variant="secondary" onClick={handleExport} loading={loading}>
      <Download className="h-4 w-4" aria-hidden="true" />
      Last ned mine data
    </Button>
  )
}
```

```typescript
'use client'

import { useState } from 'react'
import { deleteUserAccount } from '@/actions/account'
import { Button } from '@/components/ui/Button'
import { toast } from 'sonner'
import { getAuth } from 'firebase/auth'
import { app } from '@/lib/firebase'

export function DeleteAccountSection() {
  const [confirming, setConfirming] = useState(false)
  const [loading, setLoading] = useState(false)

  async function handleDelete() {
    setLoading(true)
    try {
      // Step 1: Delete Firebase Auth record client-side
      // (adminAuth is null on the server — cannot use server-side deletion)
      const auth = getAuth(app)
      const user = auth.currentUser
      if (user) {
        try {
          await user.delete()
        } catch (authError: unknown) {
          // If requires-recent-login, inform user
          const errorCode = (authError as { code?: string })?.code
          if (errorCode === 'auth/requires-recent-login') {
            toast.error('Du ma logge inn pa nytt for a slette kontoen. Logg ut og inn igjen.')
            setLoading(false)
            setConfirming(false)
            return
          }
          // Other auth errors — continue with data deletion anyway
          console.error('Firebase Auth delete failed:', authError)
        }
      }

      // Step 2: Anonymize Firestore data + delete user doc + clear session
      const result = await deleteUserAccount()
      if (!result.success) {
        toast.error(result.error || 'Kunne ikke slette kontoen.')
        setLoading(false)
        return
      }

      // Step 3: Redirect to home (session is already cleared by server action)
      toast.success('Kontoen din er slettet.')
      window.location.href = '/'
    } catch {
      toast.error('Noe gikk galt. Kontakt oss pa post@rootsculture.no.')
      setLoading(false)
    }
  }

  if (confirming) {
    return (
      <div
        role="alert"
        className="rounded-xl border border-rust/30 bg-rust/5 p-4 space-y-3"
      >
        <p className="font-medium text-forest">
          Er du sikker pa at du vil slette kontoen?
        </p>
        <p className="text-label text-body">
          Alle dine personopplysninger slettes permanent. Ordrehistorikk
          anonymiseres i henhold til regnskapsloven. Denne handlingen kan ikke
          angres.
        </p>
        <div className="flex gap-3">
          <Button
            variant="secondary"
            onClick={() => setConfirming(false)}
            disabled={loading}
          >
            Avbryt
          </Button>
          <Button
            onClick={handleDelete}
            loading={loading}
            className="bg-rust text-cream hover:bg-rust/90"
          >
            Bekreft sletting
          </Button>
        </div>
      </div>
    )
  }

  return (
    <Button variant="ghost" onClick={() => setConfirming(true)} className="text-rust">
      Slett konto
    </Button>
  )
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client -> Server Action | exportUserData and deleteUserAccount receive no user input but must verify session |
| Client Firebase Auth | user.delete() requires the current authenticated user — cannot delete other users |
| Server Action -> Firestore | Batch operations scoped to uid from verified session |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-27-05 | Elevation of Privilege | deleteUserAccount | mitigate | verifySession() at top of action — uid comes from server-verified JWT session, not client input. User can only delete own account. |
| T-27-06 | Elevation of Privilege | exportUserData | mitigate | verifySession() at top of action — uid comes from server-verified JWT session. User can only export own data. |
| T-27-07 | Information Disclosure | exportUserData JSON | mitigate | Data returned only to authenticated session owner. JSON is generated server-side with session-verified uid. No other user's data can be accessed. |
| T-27-08 | Denial of Service | Repeated exportUserData calls | accept | Low risk — queries are scoped to single user's data, which is bounded. No rate limit needed for authenticated user accessing own data. |
| T-27-09 | Tampering | deleteUserAccount called without Auth deletion | accept | If client-side user.delete() fails for non-reauthentication reasons, server action still anonymizes Firestore data and clears session.
