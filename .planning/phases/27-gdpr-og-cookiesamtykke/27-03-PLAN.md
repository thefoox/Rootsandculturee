Users can download all their personal data as JSON and permanently delete their account with proper anonymization of transaction records. The deletion flow uses a 2-step client-then-server approach: client-side Firebase Auth deletion first, then server-side Firestore anonymization — ensuring no orphaned Auth records. Additionally, fix the registration flow to persist newsletterConsent to Firestore.

Purpose: GDPR Art. 15 requires providing users access to their personal data in a portable format. Art. 17 requires the ability to request erasure. Norwegian Bokforingsloven requires 5-year retention of transaction records, so orders/bookings are anonymized rather than deleted. Art. 6(1)(a) requires consent records for newsletter — the current RegisterForm collects but does not persist newsletterConsent.

```typescript
'use server'

import { verifySession } from '@/lib/dal'
import { adminDb } from '@/lib/firebase/admin'
import { deleteSession } from '@/lib/session'

/**
 * GDPR Art. 15 — Right of access / data export
 * Returns all personal data for the authenticated user as a JSON object.
 */
export async function exportUserData(): Promise<{
  success: boolean
  data?: object
  error?: string
}> {
  const session = await verifySession()
  if (!session) {
    return { success: false, error: 'Du er ikke logget inn.' }
  }

  const uid = session.uid

  try {
    const [userDoc, ordersSnap, bookingsSnap] = await Promise.all([
      adminDb.collection('users').doc(uid).get(),
      adminDb.collection('orders').where('customerId', '==', uid).get(),
      adminDb.collection('bookings').where('customerId', '==', uid).get(),
    ])

    const data = {
      profil: userDoc.exists ? userDoc.data() : null,
      ordrer: ordersSnap.docs.map((d) => ({ id: d.id, ...d.data() })),
      bookinger: bookingsSnap.docs.map((d) => ({ id: d.id, ...d.data() })),
      eksportertDato: new Date().toISOString(),
    }

    return { success: true, data }
  } catch {
    return {
      success: false,
      error: 'Kunne ikke hente dine data. Prov igjen.',
    }
  }
}

/**
 * GDPR Art. 17 — Right to erasure
 * Anonymizes orders/bookings (Bokforingsloven: 5 year retention),
 * deletes user profile document, and clears session.
 *
 * IMPORTANT: Firebase Auth record deletion must happen CLIENT-SIDE
 * before calling this action, using auth.currentUser.delete() from
 * @/lib/firebase/client. adminAuth is null in this project.
 *
 * The client MUST:
 * 1. Call auth.currentUser.delete() first
 * 2. If it throws auth/requires-recent-login: show re-auth message, ABORT
 * 3. If it throws any other error: show error toast, ABORT (do NOT call this action)
 * 4. Only call this action after user.delete() succeeds
 */
export async function deleteUserAccount(): Promise<{
  success: boolean
  error?: string
}> {
  const session = await verifySession()
  if (!session) {
    return { success: false, error: 'Du er ikke logget inn.' }
  }

  const uid = session.uid

  try {
    // Fetch orders and bookings linked to this user
    const [ordersSnap, bookingsSnap] = await Promise.all([
      adminDb.collection('orders').where('customerId', '==', uid).get(),
      adminDb.collection('bookings').where('customerId', '==', uid).get(),
    ])

    const batch = adminDb.batch()

    // Anonymize orders — keep for Bokforingsloven (5 year retention)
    for (const doc of ordersSnap.docs) {
      batch.update(doc.ref, {
        customerId: null,
        customerEmail: '[slettet]',
        customerName: '[slettet]',
      })
    }

    // Anonymize bookings similarly
    for (const doc of bookingsSnap.docs) {
      batch.update(doc.ref, {
        customerId: null,
        customerEmail: '[slettet]',
        customerName: '[slettet]',
      })
    }

    // Delete user profile document
    batch.delete(adminDb.collection('users').doc(uid))

    await batch.commit()

    // Clear session cookie
    await deleteSession()

    return { success: true }
  } catch {
    return {
      success: false,
      error: 'Kunne ikke slette kontoen. Prov igjen eller kontakt oss.',
    }
  }
}
```

'use client'

import { useState } from 'react'
import { exportUserData } from '@/actions/account'
import { Button } from '@/components/ui/Button'
import { toast } from 'sonner'
import { Download } from 'lucide-react'

export function DataExportButton() {
  const [loading, setLoading] = useState(false)

  async function handleExport() {
    setLoading(true)
    try {
      const result = await exportUserData()
      if (!result.success || !result.data) {
        toast.error(result.error || 'Kunne ikke hente data.')
        return
      }

      // Trigger JSON file download
      const blob = new Blob([JSON.stringify(result.data, null, 2)], {
        type: 'application/json',
      })
      const url = URL.createObjectURL(blob)
      const a = document.createElement('a')
      a.href = url
      a.download = `mine-data-${new Date().toISOString().split('T')[0]}.json`
      document.body.appendChild(a)
      a.click()
      document.body.removeChild(a)
      URL.revokeObjectURL(url)
      toast.success('Data eksportert')
    } catch {
      toast.error('Noe gikk galt. Prov igjen.')
    } finally {
      setLoading(false)
    }
  }

  return (
    <Button variant="secondary" onClick={handleExport} loading={loading}>
      <Download className="h-4 w-4" aria-hidden="true" />
      Last ned mine data
    </Button>
  )
}

2. Create `src/components/konto/DeleteAccountSection.tsx`:

CRITICAL: This component implements a strict 2-step deletion flow:
- Step 1: Delete Firebase Auth record CLIENT-SIDE first
- Step 2: Call server action for Firestore anonymization ONLY if Step 1 succeeds
- If Step 1 fails with `auth/requires-recent-login`: show re-auth message, ABORT entirely
- If Step 1 fails with ANY OTHER error: show error toast, ABORT entirely (do NOT call server action)
- This prevents orphaned Auth records that could allow re-login after "deletion"

```typescript
'use client'

import { useState } from 'react'
import { deleteUserAccount } from '@/actions/account'
import { Button } from '@/components/ui/Button'
import { toast } from 'sonner'
import { auth } from '@/lib/firebase/client'

export function DeleteAccountSection() {
  const [confirming, setConfirming] = useState(false)
  const [loading, setLoading] = useState(false)

  async function handleDelete() {
    setLoading(true)
    try {
      // Step 1: Delete Firebase Auth record client-side FIRST
      // adminAuth is null on the server — cannot use server-side deletion
      const user = auth?.currentUser
      if (user) {
        try {
          await user.delete()
        } catch (authError: unknown) {
          const errorCode = (authError as { code?: string })?.code
          if (errorCode === 'auth/requires-recent-login') {
            toast.error('Du ma logge inn pa nytt for a slette kontoen. Logg ut og inn igjen.')
          } else {
            // Any other Auth error — ABORT, do NOT proceed to server action
            toast.error('Kunne ikke slette kontoen. Prov igjen senere.')
            console.error('Firebase Auth delete failed:', authError)
          }
          setLoading(false)
          setConfirming(false)
          return  // ABORT — do not call deleteUserAccount
        }
      }

      // Step 2: Auth deletion succeeded — now anonymize Firestore data + clear session
      const result = await deleteUserAccount()
      if (!result.success) {
        toast.error(result.error || 'Kunne ikke slette kontoen.')
        setLoading(false)
        return
      }

      // Step 3: Redirect to home (session is already cleared by server action)
      toast.success('Kontoen din er slettet.')
      window.location.href = '/'
    } catch {
      toast.error('Noe gikk galt. Kontakt oss pa post@rootsculture.no.')
      setLoading(false)
    }
  }

  if (confirming) {
    return (
      <div
        role="alert"
        className="rounded-xl border border-rust/30 bg-rust/5 p-4 space-y-3"
      >
        <p className="font-medium text-forest">
          Er du sikker pa at du vil slette kontoen?
        </p>
        <p className="text-label text-body">
          Alle dine personopplysninger slettes permanent. Ordrehistorikk
          anonymiseres i henhold til regnskapsloven. Denne handlingen kan ikke
          angres.
        </p>
        <div className="flex gap-3">
          <Button
            variant="secondary"
            onClick={() => setConfirming(false)}
            disabled={loading}
          >
            Avbryt
          </Button>
          <Button
            onClick={handleDelete}
            loading={loading}
            className="bg-rust text-cream hover:bg-rust/90"
          >
            Bekreft sletting
          </Button>
        </div>
      </div>
    )
  }

  return (
    <Button variant="ghost" onClick={() => setConfirming(true)} className="text-rust">
      Slett konto
    </Button>
  )
}
```
